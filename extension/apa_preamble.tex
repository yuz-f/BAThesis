% --- APA heading styles (titlesec) --------------------------
\usepackage{titlesec}

% Level 1: bold, centred, no section number prefix
\titleformat{\section}
  {\normalfont\bfseries\centering}
  {}{0em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}

% Level 2: bold, flush left
\titleformat{\subsection}
  {\normalfont\bfseries}
  {}{0em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}

% Level 3: bold italic, flush left
\titleformat{\subsubsection}
  {\normalfont\bfseries\itshape}
  {}{0em}{}
\titlespacing*{\subsubsection}{0pt}{12pt}{6pt}

% Level 4: bold, indented, inline (period after heading)
\titleformat{\paragraph}[runin]
  {\normalfont\bfseries}
  {}{0.5in}{}[.]
\titlespacing*{\paragraph}{0.5in}{12pt}{0.5em}

% Level 5: bold italic, indented, inline
\titleformat{\subparagraph}[runin]
  {\normalfont\bfseries\itshape}
  {}{0.5in}{}[.]
\titlespacing*{\subparagraph}{0.5in}{12pt}{0.5em}
