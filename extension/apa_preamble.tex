% APA 7th Edition — LaTeX preamble for Quarto PDF
% Assumes XeLaTeX (set via pdf-engine: xelatex in YAML)

% --- Title page: force pagebreak after \maketitle -----------
\usepackage{etoolbox}
\apptocmd{\maketitle}{\newpage}{}{}

% --- Abstract environment: centred heading, no indent -------
\usepackage{abstract}
\renewcommand{\abstractnamefont}{\normalfont\bfseries\centering}
\renewcommand{\abstracttextfont}{\normalfont}
\setlength{\absleftindent}{0pt}
\setlength{\absrightindent}{0pt}

% --- Paragraph indentation (0.5 in = 1.27 cm) ---------------
\setlength{\parindent}{0.5in}
\setlength{\parskip}{0pt}

% --- Running head: page number top-right, no rule -----------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\rhead{\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \rhead{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% --- APA heading styles (titlesec) --------------------------
\usepackage{titlesec}

% Level 1: bold, centred, no section number prefix
\titleformat{\section}
  {\normalfont\bfseries\centering}
  {}{0em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}

% Level 2: bold, flush left
\titleformat{\subsection}
  {\normalfont\bfseries}
  {}{0em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}

% Level 3: bold italic, flush left
\titleformat{\subsubsection}
  {\normalfont\bfseries\itshape}
  {}{0em}{}
\titlespacing*{\subsubsection}{0pt}{12pt}{6pt}

% Level 4: bold, indented, inline (period after heading)
\titleformat{\paragraph}[runin]
  {\normalfont\bfseries}
  {}{0.5in}{}[.]
\titlespacing*{\paragraph}{0.5in}{12pt}{0.5em}

% Level 5: bold italic, indented, inline
\titleformat{\subparagraph}[runin]
  {\normalfont\bfseries\itshape}
  {}{0.5in}{}[.]
\titlespacing*{\subparagraph}{0.5in}{12pt}{0.5em}

% --- Tables: only horizontal rules (booktabs) ---------------
\usepackage{booktabs}

% --- References: hanging indent (0.5 in) --------------------
% \cslhangindent is defined by Quarto's default template — only set the length
\AtBeginDocument{%
  \setlength{\cslhangindent}{0.5in}%
}

% --- No widow / orphan lines --------------------------------
\widowpenalty=10000
\clubpenalty=10000

% --- Code blocks: wrap long lines (fvextra extends fancyvrb) ----
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaksymbolleft={},
  breaklines,
  commandchars=\\\{\}
}

% --- TikZ diagrams ------------------------------------------
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, shapes.geometric, fit, backgrounds}

% --- Figures: caption label in italic (APA style) -----------
\usepackage{caption}
\DeclareCaptionLabelFormat{apa}{\textit{#1 #2}}
\captionsetup{
  labelformat=apa,
  labelsep=newline,
  justification=raggedright,
  singlelinecheck=false,
  font=normalsize
}
